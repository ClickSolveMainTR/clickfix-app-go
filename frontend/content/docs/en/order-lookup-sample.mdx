This sample is a small ClickFix app that shows the customer's orders in the conversation sidebar. It runs on Node.js 22+, has no dependencies and is MIT licensed. Copy the files below with the same folder layout and use them as the starting point of your own app. Protocol details: [Build your own app](/docs/apps?lang=en).

## What it shows [#what-it-shows]

- The `clickfix.ready` → `clickfix.context` handshake and context token renewal (`public/app.js`)
- Verifying the token on the server with the JWKS: EdDSA signature, `iss`, `aud`, `exp`/`nbf` (`lib/context-token.mjs`)
- Reading the conversation's customer email with the public API (`server.mjs`)
- Adding an internal note and replying to the customer with `clickfix.action`
- Light/dark theme, a layout that fits the narrow sidebar, and framing allowed only for the ClickFix panel (`frame-ancestors`)

## Run it [#run]

Put the files in a folder, create `data/orders.json` and start the server:

```bash
CLICKFIX_APP_ID=<app-id> \
CLICKFIX_API_KEY=cf_sk_... \
PANEL_ORIGINS=https://getclickfix.com \
npm start
```

| Variable           | Description                                                                                                                                           |
| ------------------ | ----------------------------------------------------------------------------------------------------------------------------------------------------- |
| `CLICKFIX_APP_ID`  | The **App ID** shown on the app's card after you register it in ClickFix. It is the token's `aud`.                                                    |
| `CLICKFIX_API_KEY` | An [API key](/docs/api?lang=en) with the `conversations:read` scope. Without it the app runs in demo mode and shows the orders of `demo@example.com`. |
| `CLICKFIX_API_URL` | Default `https://api.getclickfix.com`                                                                                                                 |
| `PANEL_ORIGINS`    | Panel origin(s) allowed to frame the app, comma separated.                                                                                            |
| `PORT`             | Default `8787`                                                                                                                                        |

The app must be served over HTTPS from a **different origin** than the ClickFix panel (e.g. `https://orders.example.com`). To try it locally, use an HTTPS tunnel (Cloudflare Tunnel, ngrok, etc.).

## Add it to ClickFix [#add-to-clickfix]

1. On **Organization → Manage apps**, click **Install** on the **Order lookup** card in the gallery.
2. Enter your app's HTTPS URL and save.
3. Copy the **App ID** shown on the card and pass it as `CLICKFIX_APP_ID`.
4. Open a conversation; the orders appear under **Apps** in the right panel.

```js tab="server.mjs"
// ClickFix example app: order lookup for the conversation sidebar.
//
//   node server.mjs            (Node 22+, no dependencies)
//
// Environment:
//   CLICKFIX_APP_ID     App id from ClickFix (Uygulamalar → app → "Uygulama kimliği"). Required.
//   CLICKFIX_API_URL    ClickFix API origin. Default https://api.getclickfix.com
//   CLICKFIX_API_KEY    API key with conversations:read. Without it the app runs in demo mode.
//   PANEL_ORIGINS       Comma-separated origins allowed to frame the app. Default https://getclickfix.com
//   PORT                Default 8787
import { createServer } from "node:http";
import { readFile } from "node:fs/promises";
import { extname, join, normalize } from "node:path";
import { fileURLToPath } from "node:url";

import { keysFromJWKS, verifyContextToken } from "./lib/context-token.mjs";

const root = fileURLToPath(new URL(".", import.meta.url));
const API = (
  process.env.CLICKFIX_API_URL || "https://api.getclickfix.com"
).replace(/\/$/, "");
const APP_ID = process.env.CLICKFIX_APP_ID || "";
const API_KEY = process.env.CLICKFIX_API_KEY || "";
const PANEL_ORIGINS = (process.env.PANEL_ORIGINS || "https://getclickfix.com")
  .split(",")
  .map((o) => o.trim())
  .filter(Boolean);
const PORT = Number(process.env.PORT || 8787);
const orders = JSON.parse(
  await readFile(join(root, "data/orders.json"), "utf8"),
);

const TYPES = {
  ".html": "text/html; charset=utf-8",
  ".js": "text/javascript; charset=utf-8",
  ".css": "text/css; charset=utf-8",
  ".svg": "image/svg+xml",
};

let jwks = { keys: new Map(), at: 0 };
async function signingKeys() {
  if (Date.now() - jwks.at < 10 * 60_000 && jwks.keys.size) return jwks.keys;
  const res = await fetch(`${API}/v1/app-context/jwks`, {
    signal: AbortSignal.timeout(10_000),
  });
  if (!res.ok) throw new Error(`jwks ${res.status}`);
  jwks = { keys: keysFromJWKS(await res.json()), at: Date.now() };
  return jwks.keys;
}

/** The customer's email for the conversation, read with the public API. */
async function contactEmail(conversationUUID) {
  const res = await fetch(
    `${API}/v1/public-api/conversations/${encodeURIComponent(conversationUUID)}`,
    {
      headers: { Authorization: `Bearer ${API_KEY}` },
      signal: AbortSignal.timeout(10_000),
    },
  );
  if (!res.ok) throw new Error(`conversation ${res.status}`);
  const body = await res.json();
  return body?.data?.contact?.email ?? null;
}

function send(res, status, body, headers = {}) {
  res.writeHead(status, {
    "Content-Type": "application/json; charset=utf-8",
    "Cache-Control": "no-store",
    ...headers,
  });
  res.end(JSON.stringify(body));
}

async function lookup(req, res) {
  const token = (req.headers.authorization || "").replace(/^Bearer /, "");
  let keys;
  try {
    keys = await signingKeys();
  } catch {
    return send(res, 503, { error: "jwks_unavailable" });
  }
  let claims;
  try {
    claims = verifyContextToken(token, { keys, appId: APP_ID });
  } catch (err) {
    return send(res, 401, {
      error: "invalid_context",
      detail: String(err.message),
    });
  }
  if (!claims.conversation_uuid)
    return send(res, 200, {
      email: null,
      orders: [],
      reason: "no_conversation",
    });
  try {
    const email = API_KEY
      ? await contactEmail(claims.conversation_uuid)
      : "demo@example.com";
    const found = email
      ? orders.filter((o) => o.email.toLowerCase() === email.toLowerCase())
      : [];
    return send(res, 200, { email, demo: !API_KEY, orders: found });
  } catch {
    return send(res, 502, { error: "lookup_failed" });
  }
}

const server = createServer(async (req, res) => {
  const url = new URL(req.url, "http://localhost");
  if (req.method === "GET" && url.pathname === "/api/orders")
    return lookup(req, res);
  // The page posts `clickfix.ready` only to these origins and accepts messages only from them.
  if (req.method === "GET" && url.pathname === "/api/config")
    return send(res, 200, { panel_origins: PANEL_ORIGINS });
  if (req.method !== "GET")
    return send(res, 405, { error: "method_not_allowed" });

  const rel = normalize(
    url.pathname === "/" ? "/index.html" : url.pathname,
  ).replace(/^(\.\.[/\\])+/, "");
  const file = join(root, "public", rel);
  if (!file.startsWith(join(root, "public")))
    return send(res, 404, { error: "not_found" });
  try {
    const body = await readFile(file);
    res.writeHead(200, {
      "Content-Type": TYPES[extname(file)] || "application/octet-stream",
      // Only the ClickFix panel may frame this app.
      "Content-Security-Policy": `default-src 'self'; frame-ancestors ${PANEL_ORIGINS.join(" ")}`,
      "Referrer-Policy": "no-referrer",
      "X-Content-Type-Options": "nosniff",
    });
    res.end(body);
  } catch {
    send(res, 404, { error: "not_found" });
  }
});

if (!APP_ID)
  console.warn(
    "CLICKFIX_APP_ID is not set: every context token will be rejected.",
  );
if (!API_KEY)
  console.warn(
    "CLICKFIX_API_KEY is not set: demo mode, showing orders of demo@example.com.",
  );
server.listen(PORT, () => console.log(`order-lookup listening on :${PORT}`));
```

```js tab="lib/context-token.mjs"
// Verifies the ClickFix app context token (EdDSA JWT) without dependencies.
//
// The panel posts the token to the iframe in a `clickfix.context` message.
// Your server must check: signature against the JWKS published at
// GET {api}/v1/app-context/jwks, alg=EdDSA, iss=clickfix:sidebar-apps,
// aud=your app's id, and the nbf/exp window. Only then trust the claims.
import { createPublicKey, verify } from "node:crypto";

export const ISSUER = "clickfix:sidebar-apps";
const SKEW_SECONDS = 30;

function b64url(part) {
  return Buffer.from(part, "base64url");
}

/** Turns the JWKS envelope (`{success, data: {keys}}` or `{keys}`) into kid → KeyObject. */
export function keysFromJWKS(body) {
  const keys = body?.data?.keys ?? body?.keys ?? [];
  const out = new Map();
  for (const jwk of keys) {
    if (jwk.kty !== "OKP" || jwk.crv !== "Ed25519") continue;
    out.set(
      jwk.kid,
      createPublicKey({
        key: { kty: "OKP", crv: "Ed25519", x: jwk.x },
        format: "jwk",
      }),
    );
  }
  return out;
}

/**
 * Returns the verified claims or throws. `keys` comes from keysFromJWKS,
 * `appId` is the app's id shown in ClickFix (the token audience).
 */
export function verifyContextToken(token, { keys, appId, now = Date.now() }) {
  if (typeof token !== "string") throw new Error("token missing");
  const parts = token.split(".");
  if (parts.length !== 3) throw new Error("malformed token");
  const [h, p, s] = parts;
  const header = JSON.parse(b64url(h).toString("utf8"));
  if (header.alg !== "EdDSA") throw new Error("unexpected alg");
  const key =
    keys.get(header.kid) ??
    (keys.size === 1 ? [...keys.values()][0] : undefined);
  if (!key) throw new Error("unknown key");
  if (!verify(null, Buffer.from(`${h}.${p}`), key, b64url(s)))
    throw new Error("bad signature");

  const claims = JSON.parse(b64url(p).toString("utf8"));
  const t = Math.floor(now / 1000);
  const aud = Array.isArray(claims.aud) ? claims.aud : [claims.aud];
  if (claims.iss !== ISSUER) throw new Error("unexpected issuer");
  if (!appId || !aud.includes(appId)) throw new Error("unexpected audience");
  if (typeof claims.exp !== "number" || claims.exp + SKEW_SECONDS < t)
    throw new Error("token expired");
  if (typeof claims.nbf === "number" && claims.nbf - SKEW_SECONDS > t)
    throw new Error("token not yet valid");
  return claims;
}
```

```js tab="public/app.js"
// Browser side of the ClickFix app protocol:
//   1. app → panel   {type: "clickfix.ready"}
//   2. panel → app   {type: "clickfix.context", context_token, scopes}   (again on refresh, ~every 100 s)
//   3. app → panel   {type: "clickfix.action", request_id, action, content | value | target_uuid}
//   4. panel → app   {type: "clickfix.action.result", request_id, success, error?}
// The token is only sent to this app's own server; never put it in a URL or storage.

const $ = (id) => document.getElementById(id);
const fmtMoney = (n, c) =>
  new Intl.NumberFormat("tr-TR", { style: "currency", currency: c }).format(n);
const fmtDate = (iso) =>
  new Intl.DateTimeFormat("tr-TR", { dateStyle: "medium" }).format(
    new Date(iso),
  );
const STATUS = {
  processing: "Hazırlanıyor",
  shipped: "Kargoda",
  delivered: "Teslim edildi",
  refunded: "İade edildi",
};

const { panel_origins: origins } = await (await fetch("/api/config")).json();
let ctx = null;
let lastToken = null;
const pending = new Map();

function post(message) {
  // A wrong targetOrigin is dropped by the browser, so posting to each allowed origin is safe.
  for (const origin of origins) window.parent.postMessage(message, origin);
}

function act(action, content) {
  if (!ctx?.scopes.includes(action))
    return Promise.resolve({ success: false, error: "scope_denied" });
  const request_id = crypto.randomUUID();
  return new Promise((resolve) => {
    pending.set(request_id, resolve);
    post({ type: "clickfix.action", request_id, action, content });
  });
}

function summary(o) {
  const lines = [
    `Sipariş ${o.id}: ${STATUS[o.status] ?? o.status}`,
    `Tutar: ${fmtMoney(o.total, o.currency)}`,
    `Tarih: ${fmtDate(o.placed_at)}`,
  ];
  if (o.tracking) lines.push(`Kargo: ${o.carrier} · ${o.tracking}`);
  return lines.join("\n");
}

function render(data) {
  const list = $("orders");
  list.replaceChildren();
  $("who").textContent = data.email
    ? `${data.email}${data.demo ? " (demo verisi)" : ""}`
    : "Konuşmanın e-posta adresi yok.";
  if (data.reason === "no_conversation") {
    $("status").textContent =
      "Siparişleri görmek için uygulamayı bir konuşmanın kenar çubuğunda açın.";
    return;
  }
  $("status").textContent = data.orders.length
    ? ""
    : "Bu müşteriye ait sipariş bulunamadı.";
  for (const o of data.orders) {
    const li = $("order").content.firstElementChild.cloneNode(true);
    li.querySelector(".id").textContent = o.id;
    li.querySelector(".badge").textContent = STATUS[o.status] ?? o.status;
    li.querySelector(".meta").textContent =
      `${fmtDate(o.placed_at)} · ${fmtMoney(o.total, o.currency)}`;
    li.querySelector(".items").textContent = o.items.join(", ");
    for (const [cls, action] of [
      ["note", "conversation.note"],
      ["reply", "conversation.reply"],
    ]) {
      const btn = li.querySelector(`.${cls}`);
      btn.hidden = !ctx.scopes.includes(action);
      btn.addEventListener("click", async () => {
        btn.disabled = true;
        const result = await act(action, summary(o));
        btn.disabled = false;
        $("status").textContent = result.success
          ? "Tamam."
          : `Yapılamadı (${result.error ?? "bilinmeyen hata"}).`;
      });
    }
    list.append(li);
  }
}

async function load() {
  $("status").textContent = "Siparişler yükleniyor…";
  const res = await fetch("/api/orders", {
    headers: { Authorization: `Bearer ${ctx.token}` },
  });
  if (!res.ok) {
    $("status").textContent =
      res.status === 401
        ? "Bağlam doğrulanamadı. Paneli yenileyin."
        : "Siparişler yüklenemedi. Tekrar deneyin.";
    return;
  }
  render(await res.json());
}

window.addEventListener("message", (event) => {
  if (event.source !== window.parent || !origins.includes(event.origin)) return;
  const msg = event.data;
  if (msg?.type === "clickfix.context") {
    const first = lastToken === null;
    ctx = { token: msg.context_token, scopes: msg.scopes ?? [] };
    lastToken = msg.context_token;
    // A refresh keeps the same conversation (the panel remounts the frame on a change).
    if (first) void load();
  } else if (msg?.type === "clickfix.action.result") {
    pending.get(msg.request_id)?.({ success: msg.success, error: msg.error });
    pending.delete(msg.request_id);
  }
});

post({ type: "clickfix.ready" });
```
